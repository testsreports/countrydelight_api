function finalGenerator(skuObj) {

            "force_scanned_count": 0,
            "manual_quantity": 0,
            "manual_weight": 0,
            "product_id": item.productId,

            "weight": item.quantity * item.weight
        };
        results.push(resultEntry); // Pushes each product Object in result array

}

//function to generate random number
const characters = '0123456789';
function generateString(length) {
    let result = '';
    for (let i = 0; i < length; i++) {
        result += characters.charAt(Math.floor(Math.random() * 10));
    }
    return result;
}

function fromBaseN(baseNString, base, baseDigits) {
    if (base > baseDigits.length) {
        throw new Error("Base must not exceed the length of the digit set.");
    }
    let output = 0n;
    const reversed = baseNString.split('').reverse().join('');

    for (let powerNum = 0; powerNum < reversed.length; powerNum++) {
        const character = reversed[powerNum];
        const index = baseDigits.indexOf(character);
        if (index === -1) {
            throw new Error(`Character '${character}' not found in base digits.`);
        }
        output += BigInt(index) * BigInt(base) ** BigInt(powerNum);
    }
    return output.toString();
}
function toBaseN(decimalNumber, base, baseDigits) {
    if (base > baseDigits.length) {
        throw new Error("Base must not exceed the length of the digit set.");
    }
    if (decimalNumber === '0') {
        return "0";
    }
    let num = BigInt(decimalNumber);
    let baseN = '';

    while (num > 0n) {
        const remainder = Number(num % BigInt(base));
        baseN = baseDigits[remainder] + baseN;
        num /= BigInt(base);
    }
    return baseN;
}

function getCurrentTimeStamp(reqFormat,days) {
    const moment = require('moment');
    reqDate = moment().add(days, 'days').format(reqFormat);
    return reqDate;
}
