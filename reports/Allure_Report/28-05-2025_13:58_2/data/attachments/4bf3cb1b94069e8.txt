try{
    const jsonData = pm.response.json();
    pm.test("Printing Crate QR", function () {
     pm.collectionVariables.set("crate_QR", jsonData[0].qr_code);
     console.log(jsonData[0].qr_code)
    });

    pm.test("Printing Crate Type", function () {
     if(jsonData[0].tag == 0){
        console.log("Regular Old Format Crate")
     }
     else if(jsonData[0].tag == 1){
        console.log("Regular New Format Crate")
     }
     else if(jsonData[0].tag == 2){
        console.log("Bad Quality Crate")
     }
     else if(jsonData[0].tag == 3){
        console.log("Expired Item Crate")
     }
     else{
        console.log(`Something went wrong, The crate tag in response is ${jsonData[0].tag}, which is not belongs to any category`)
     }
    });
}
catch(err){
    console.log(err)
    if(err){
        pm.test("Unsuccessful API response", function () {
            pm.response.not.to.have.status(201);
});
    }
}


pm.test("Successful POST request", function () {
    pm.expect(pm.response.code).to.be.oneOf([201, 202]);
});
