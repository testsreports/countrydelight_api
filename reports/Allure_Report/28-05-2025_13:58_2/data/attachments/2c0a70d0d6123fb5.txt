function finalGenerator(skuObj) {
    var results = []; // Variable to store final result or QRs

    for (let i = 0; i < 1; i++) {
        var item = skuObj[i];
        var resultEntry = {
            "force_scanned_count": 0,
            "manual_quantity": 0,
            "manual_weight": 0,
            "product_id": item.productId,
            "qr_codes": qrCodeGenerator(item.sku, item.weight, item.quantity),
            "weight": Math.floor((item.quantity * item.weight))
        };
        results.push(resultEntry); // Pushes each product Object in result array
    };
    return results; // Return the final result
}

function qrCodeGenerator(sku, weight, quantity) {
    var qrCodes = [];

    for (var i = 0; i < quantity; i++) {

        //In case of current date variable "getCurrentDate()" can be used, In case of other else date, need to pass date
        var flag = sku + "#" + getCurrentDate() + "#" + weight + "#" + '02#'.concat(generateString(8));
        qrCodes.push(flag);
        //console.log(flag)
    }
    return qrCodes;
}

//function to generate random number
const characters = '0123456789';
function generateString(length) {
    let result = '';
    for (let i = 0; i < length; i++) {
        result += characters.charAt(Math.floor(Math.random() * 10));
    }
    return result;
}
