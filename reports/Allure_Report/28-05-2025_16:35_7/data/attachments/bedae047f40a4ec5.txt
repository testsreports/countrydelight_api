const response = pm.response.json();

try{
    if (pm.response.code === 200) {
    pm.collectionVariables.set("requestId", response.id);
    console.info(`Request ID received is ${response.id}`)
    }
}
catch(err){
    console.error(err)
}


// @allure.label.parentSuite=api
// @allure.label.suite=response
// @allure.label.subSuite=time
// @allure.label.tag=responseSuccess

pm.test("✅ Response Status is 200", function () {
    pm.expect(pm.response.code).to.eql(200);
});

pm.test("✅ OTP successfully sent to Mobile", function () {
    pm.expect(pm.response.text()).to.include("OTP has been sent to the mobile number.");
});

pm.test("✅ Valid request ID", function(){
    pm.expect(response.id).to.not.be.null;
    pm.expect(response.id).to.not.be.undefined;
    pm.expect(response.id).to.be.a('number');
});


pm.test("✅ Response 'data' is present", function () {
    pm.expect(response.data).to.not.eql(null);
});

// @allure.label.suite=responseTime
// @allure.label.tag=responseTime
pm.test("✅ Response time is under 1000ms", function () {
    pm.expect(pm.response.responseTime).to.be.below(3000);
});
