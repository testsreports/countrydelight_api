function finalGenerator(skuObj) {
    var results = []; // Variable to store final result

    for (let i = 0; i < no_of_diff_SKU_in_crate; i++) {
        var item = skuObj[i];
        var resultEntry = {
            "force_scanned_count": 0,
            "manual_quantity": 0,
            "manual_weight": 0,
            "product_id": item.productId,
            "qr_codes": qrCodeGenerator(item.sku, item.weight, item.quantity),
            "weight": item.quantity * item.weight
        };
        results.push(resultEntry); // Pushes each product Object in result array
        //console.log(resultEntry)
    };
    return results; // Return the final result
}

function qrCodeGenerator(sku, weight, quantity) {
    var qrCodes = [];

    for (var i = 0; i < quantity; i++) {
        var flag = sku + "#" + getCurrentDate() + "#" + weight + "#" + '2312#'.concat(generateString(8));
        qrCodes.push(flag);
    }

    return qrCodes;
}

const characters = '0123456789';
function generateString(length) {
    let result = '';
    for (let i = 0; i < length; i++) {
        result += characters.charAt(Math.floor(Math.random() * 10));
    }

    return result;
}

pm.collectionVariables.set("QRobjectDCenter", JSON.stringify(finalGenerator(skuObj)));
